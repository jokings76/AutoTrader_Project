    def _get_merged_candles(
        self, stock_code: str, interval: int = 1, count: int = 60
    ) -> list:
        """
        ?뱀씪 遺꾨큺留뚯쑝濡?媛쒖닔媛 遺議깊븷 寃쎌슦, ?꾩씪 遺꾨큺???뚯뼱??遺숈뿬??媛쒖닔瑜?留욎땄.

        """
        # 1. ?쇰떒 ?뱀씪 ?곗씠???붿껌
        today_candles = self.api.get_minute_candles(
            stock_code, interval=interval, count=count
        )

        # 2. ?붽뎄 媛쒖닔瑜?梨꾩썱嫄곕굹 ?뱀씪 ?곗씠?곌? ?꾩삁 ?놁쑝硫?洹몃?濡?諛섑솚
        if not today_candles or len(today_candles) >= count:
            return today_candles or []

        # 3. 遺議깊븳 媛쒖닔 怨꾩궛
        needed = count - len(today_candles)

        # 4. ?꾩씪 ?좎쭨 怨꾩궛
        yesterday = (self._now() - timedelta(days=1)).strftime("%Y%m%d")

        try:
            # 5. ?꾩씪 ?곗씠???붿껌 (理쒓렐 遊됰???怨쇨굅濡??대젮??
            yesterday_candles = self.api.get_minute_candles(
                stock_code, interval=interval, count=needed, base_date=yesterday
            )

            if yesterday_candles:
                # 6. ?꾩씪 ?곗씠??怨쇨굅) + ?뱀씪 ?곗씠??理쒓렐) ?쒖꽌濡?蹂묓빀
                merged = yesterday_candles + today_candles
                logger.info(
                    f"?뱤 [{stock_code}] 遺꾨큺 遺議?蹂댁셿: ?뱀씪 {len(today_candles)}媛?"
                    f"+ ?꾩씪 {len(yesterday_candles)}媛?= 珥?{len(merged)}媛?

                return merged
        except Exception as e:
            logger.warning(f"[{stock_code}] ?꾩씪 遺꾨큺 議고쉶 ?ㅽ뙣: {e}")

        # ?꾩씪 議고쉶 ?ㅽ뙣 ??洹몃깷 ?뱀씪 ?곗씠?곕씪??諛섑솚
        return today_candles

    # ?쒖닔?듬쪧 怨꾩궛 (?섏닔猷?李④컧)
    # ========================================
    @staticmethod
    def _gross_rate(buy_price: float, current_price: float) -> float:
        return (current_price - buy_price) / buy_price if buy_price else 0.0

    @staticmethod
    def _net_rate(buy_price: float, current_price: float) -> float:
        """?섏닔猷??뺣났)+?멸툑 李④컧 ?쒖닔?듬쪧."""
        if not buy_price:
            return 0.0
        gross = (current_price - buy_price) / buy_price
        return gross - ROUND_TRIP_COST

    @staticmethod
    def _net_profit(buy_price: float, current_price: float, qty: int) -> float:
        """?ㅼ젣 ?쒖넀??湲덉븸 (?섏닔猷??멸툑 李④컧)."""
        buy_amt = buy_price * qty
        sell_amt = current_price * qty
        cost = buy_amt * COMMISSION_RATE + sell_amt * (COMMISSION_RATE + TAX_RATE)
        return (sell_amt - buy_amt) - cost

    def on_trade(self, parsed_trade: dict, now: float = None):
        code = parsed_trade.get("stock_code")
        if not code:
            return

        if code in self.holdings:
            price = parsed_trade.get("price")
            if price:
                self.on_price_update(code, price)
            return

        if self.phase1b and self.phase1b.is_watching(code):
            state = self.phase1b.on_trade(parsed_trade, now=now)
            if state == ChemulState.READY_TO_BUY:
                self._try_phase1b_buy(code, now)

        if self.phase3 and self.phase3.is_watching(code):
            from core.strategy.phase3_controller import Phase3State

            new_state = self.phase3.on_trade(parsed_trade)
            if new_state == Phase3State.READY_TO_BUY:
                self._try_phase3_buy(code)

    def on_orderbook(self, parsed_orderbook: dict, now: float = None):
        code = parsed_orderbook.get("stock_code")
        if not code:
            return
        if self.phase1b and self.phase1b.is_watching(code):
            state = self.phase1b.on_orderbook(parsed_orderbook, now=now)
            if state == ChemulState.READY_TO_BUY:
                self._try_phase1b_buy(code, now)
        # Phase3 B ?몃━嫄곗슜: ?멸? ?붾웾 鍮꾩쑉留?媛깆떊 (留ㅼ닔 ?먯젙? on_trade?먯꽌)
        if self.phase3 and self.phase3.is_watching(code):
            try:
                self.phase3.on_orderbook(parsed_orderbook)
            except Exception:
                logger.exception("[%s] phase3 on_orderbook ?덉쇅", code)

    def _get_market_defense_mode(self) -> str:
        """
        吏??諛⑹뼱 紐⑤뱶 ?뺤씤
        諛섑솚: "NORMAL" (-3% ?댄븯), "CAUTION" (-3% ~ -5%), "HALT" (-5% 珥덇낵)
        """
        try:
            # 肄붿뒪??吏??肄붾뱶 (肄붿뒪???곗떆硫?"0001"濡?蹂寃쏀븯?몄슂)
            index_code = "1001" 
            index_data = self.api.get_minute_candles(index_code, interval=1, count=2)

            if not index_data or len(index_data) < 2:
                return "NORMAL"

            current = float(index_data[0].get('close', 0))
            open_price = float(index_data[0].get('open', 0))

            if open_price <= 0 or current <= 0:
                return "NORMAL"

            index_change_pct = (current - open_price) / open_price * 100

            if index_change_pct <= -5.0:
                logger.warning(f"?썞 留덉폆 ?щ옒??諛쒕룞! 吏??{index_change_pct:+.2f}% - ?꾨㈃ 留ㅻℓ 以묐떒")
                return "HALT"
            elif index_change_pct <= -3.0:
                logger.info(f"?좑툘 吏??寃쎈낫 諛쒕룞! 吏??{index_change_pct:+.2f}% - 媛뺤꽭 議곌굔 媛뺥솕 (7.5??")
                return "CAUTION"
            else:
                return "NORMAL"

        except Exception as e:
            logger.debug(f"吏???뺤씤 ?먮윭: {e}")
            return "NORMAL"

    def evaluate_new_intensity_strategy(self, stock_code: str, candles: list, current_price: int, open_price: int) -> tuple[bool, dict]:
        """泥닿껐媛뺣룄 吏??& 30遊??좉퀬媛 湲곕컲???덈줈???됯? 濡쒖쭅 (吏??諛⑹뼱留??ы븿)"""
        now = self._now().time()

        # ==========================================
        # [理쒖슦?? 吏??諛⑹뼱留??뺤씤
        # ==========================================
        market_mode = self._get_market_defense_mode()

        if market_mode == "HALT":
            return False, {"reason": "吏??-5% 珥덇낵濡??명븳 ?꾨㈃ 留ㅻℓ 以묐떒", "score": 0.0}

        # 1. 30遊??좉퀬媛 ?꾪꽣 (紐⑤뱺 ?쒓컙? 怨듯넻)
        if not calc_ma.is_30_candle_new_high(candles):
            return False, {"reason": "30遺??좉퀬媛 ?꾨떂", "score": 0.0}

        # 2. ?쒓컙?蹂?泥닿껐媛뺣룄 吏???꾪꽣
        is_pass = False
        if now < PHASE1A_CUT_OFF_TIME:
            if not open_price or open_price <= 0:
                return False, {"reason": "?쒖옉媛 ?놁쓬", "score": 0.0}

            change_pct = (current_price - open_price) / open_price * 100

            if change_pct <= 5.0:

    @staticmethod
    def _volume_ratio(candles: list[dict]) -> float:
        cur_vol = candles[0]["volume"]
        prev = candles[1 : 1 + VOLUME_LOOKBACK]
        if not prev:
            return 0.0
        avg = sum(c["volume"] for c in prev) / len(prev)
        return cur_vol / avg if avg > 0 else 0.0

    def _resolve_position_amount(
        self, stock_code: str, sub_strategy: str
    ) -> tuple[int, Optional[dict]]:
        if self.optimizer is None:
            return POSITION_AMOUNT, None
        try:
            opt_info = self.optimizer.calculate_position_amount(
                stock_code, sub_strategy
            )
            amount = int(opt_info.get("amount", 0))
            if amount <= 0:
                return POSITION_AMOUNT, None
            return amount, opt_info
        except Exception:
            logger.exception(f"[{stock_code}] 鍮꾩쨷 怨꾩궛 ?ㅽ뙣, fallback")
            return POSITION_AMOUNT, None

    def _execute_buy(self, stock_code, stock_name, phase, info, sub_strategy):
        current_price = info["current_price"]
        sc = info.get("score")
        if sc is not None:
            logger.info(
                "[%s] %s 留ㅼ닔?됯? ?듦낵 | score=%.2f/%.2f | %s",
                stock_code,
                stock_name,
                sc,
                info.get("score_threshold", 0),
                info.get("score_breakdown", ""),
            )

        position_amount, opt_info = self._resolve_position_amount(
            stock_code, sub_strategy
        )
        quantity = int(position_amount // current_price)
        if quantity < 1:
            logger.warning("[%s] %s ?섎웾 0 -> skip", stock_code, stock_name)
            return

        if opt_info:
            logger.info(
                "[%s] ?숈쟻 鍮꾩쨷 %.2fx -> %s??,
                stock_code,
                opt_info.get("final_weight", 1.0),
                f"{position_amount:,}",
            )

        self.pending.add(stock_code)
        try:
            ma_val = info.get("ma5") or 0
            if sub_strategy == "1B":
                entry_reason = f"Phase 1B 泥닿껐媛뺣룄 (?꾩옱媛 {current_price:,})"
            elif sub_strategy == "3":
                entry_reason = f"Phase 3 ?⑥“利앷? 1遺꾩쑀吏 (?꾩옱媛 {current_price:,})"
            elif sub_strategy == "1S":

                    f"湲됰벑 吏꾩엯 | ?쒖큹媛+{info.get('surge_rate', 0)*100:.2f}% "
                    f"| vol x{info.get('volume_ratio', 0):.2f} (?꾩옱媛 {current_price:,})"
                )
            else:
                # 1A=5MA, 2=30MA (PHASE2_MA_PERIOD)
                ma_label = "5" if sub_strategy == "1A" else str(PHASE2_MA_PERIOD)
                entry_reason = (
                    f"Phase{sub_strategy} | "
                    f"MA{ma_label}={ma_val:,.0f} "
                    f"| vol x{info.get('volume_ratio', 0):.2f}"
                )
                if sub_strategy == "1A":
                    entry_reason += f" | ?쒖큹媛+{info.get('surge_rate', 0)*100:.2f}%"
            if opt_info:
                entry_reason += f" | 鍮꾩쨷x{opt_info.get('final_weight', 1.0):.2f}"

            if not result or not result.get("success"):

                logger.error("[%s] 留ㅼ닔 ?ㅽ뙣: %s", stock_code, err)

                    "ORDER_FAIL", f"BUY {stock_code}: {err}", "ERROR"
                )
                _notify(f"留ㅼ닔 ?ㅽ뙣\n{stock_code} {stock_name}\n?ъ쑀: {err}")
                return

            trade_id = TradeRepository.insert_buy(
                stock_code=stock_code,
                stock_name=stock_name,
                buy_price=current_price,
                buy_quantity=quantity,
                strategy_phase=phase,
                sub_strategy=sub_strategy,
                entry_reason=entry_reason,
            )

            self.holdings[stock_code] = {
                "trade_id": trade_id,
                "buy_price": current_price,
                "buy_quantity": quantity,
                "buy_time": self._now(),
                "stock_name": stock_name,
                "strategy_phase": phase,
                "sub_strategy": sub_strategy,
                "highest_price": current_price,
                "ma20": None,
                "ma20_updated": None,
                "trigger": info.get("trigger"),  # Phase3 A/B (Phase3留??명똿)
                "opening_price": info.get("opening_price", 0.0),  # B ?먯젅 湲곗???
                "position_weight": (opt_info or {}).get("final_weight", 1.0),
                "warmup_until": self._now() + BUY_WARMUP,
            }

            self._sold_at.pop(stock_code, None)
            self._buy_success_count += 1

            logger.info(
                "BUY [%s] %s %d二?@ %s??(%s) = %s??| ?뚮컢??%ds",
                stock_code,
                stock_name,
                quantity,
                f"{current_price:,}",
                sub_strategy,
                f"{current_price * quantity:,}",
                int(BUY_WARMUP.total_seconds()),
            )
            SystemEventRepository.log(
                "BUY",
                f"{stock_code} {stock_name} {quantity}二?@ {current_price:,}??[{sub_strategy}]",
                "INFO",
            )
            weight_str = f" (鍮꾩쨷 {opt_info['final_weight']:.2f}x)" if opt_info else ""
            _notify(
                f"留ㅼ닔 泥닿껐 [{sub_strategy}]\n醫낅ぉ: {stock_name} ({stock_code})\n"
                f"?섎웾: {quantity}二?@ {current_price:,}??weight_str}\n"
                f"湲덉븸: {current_price * quantity:,}??n?꾨왂: {entry_reason}"
            )
            self._mark_watch_bought(stock_code)
        finally:
            self.pending.discard(stock_code)

    def _record_watch_list(self, stock_code, stock_name, phase, info):

    def on_price_update(self, stock_code: str, current_price: float):
        pos = self.holdings.get(stock_code)
        if not pos or not current_price:
            return
        if stock_code in self.sell_blocked:
            return

        buy_price = pos["buy_price"]
        if not buy_price:
            return

        if current_price > pos["highest_price"]:
            pos["highest_price"] = current_price

        warmup_until = pos.get("warmup_until")
        if warmup_until and self._now() < warmup_until:
            return

        gross_rate = self._gross_rate(buy_price, current_price)  # 媛寃?湲곗?
        net_rate = gross_rate - ROUND_TRIP_COST  # ?쒖씡 湲곗?
        trigger = pos.get("trigger")  # Phase3 "A"/"B"/None

        exit_reason = None

        # 1) ?먯젅: B???쒓?-1%, 洹??몃뒗 媛寃?3%
        if trigger == "B":
            opening = pos.get("opening_price", 0.0)
            if opening > 0:
                stop_line = opening * (1 + B_STOP_LOSS_FROM_OPEN)  # opening * 0.99
                if current_price <= stop_line:
                    drop = (current_price - opening) / opening * 100
                    exit_reason = (
                        f"B ?먯젅 ?쒓?{B_STOP_LOSS_FROM_OPEN*100:.1f}% "
                        f"(?쒓? {opening:,.0f} ??{current_price:,}, "
                        f"{drop:+.2f}%, ??{net_rate*100:+.2f}%)"
                    )
        else:
            if gross_rate <= STOP_LOSS_RATE:
                exit_reason = f"?먯젅 媛寃?gross_rate*100:.2f}% (??{net_rate*100:.2f}%)"

        # 2) ?듭젅
        if exit_reason is None:
            buy_t = pos["buy_time"].time() if hasattr(pos["buy_time"], "time") else None

            if trigger in ("A", "B"):
                # Phase 3 (A쨌B ?????몃젅?쇰쭅)
                highest = pos.get("highest_price", buy_price)
                peak_net = self._net_rate(buy_price, highest)
                if peak_net >= TRAIL_ACTIVATE:
                    pos["trail_armed"] = True
                if pos.get("trail_armed"):
                    trail_line = highest * (1 - TRAIL_GIVEBACK)
                    if current_price <= trail_line:
                        give = (current_price - highest) / highest * 100
                        exit_reason = (
                            f"?몃젅?쇰쭅 怨좎젏-{TRAIL_GIVEBACK*100:.0f}% "
                            f"({give:+.2f}%, ??{net_rate*100:+.2f}%) [{trigger}]"
                        )
            else:
                # 湲곗〈 1A/1B/2/1S ?? 留ㅼ닔 ?쒓컖?쇰줈 ?몃젅?쇰쭅 or ?듭젅罹?
                use_trailing = buy_t is not None and buy_t < TRAIL_BUY_CUTOFF
                if use_trailing:
                    highest = pos.get("highest_price", buy_price)
                    peak_net = self._net_rate(buy_price, highest)
                    if peak_net >= TRAIL_ACTIVATE:
                        pos["trail_armed"] = True
                    if pos.get("trail_armed"):
                        trail_line = highest * (1 - TRAIL_GIVEBACK)
                        if current_price <= trail_line:
                            give = (current_price - highest) / highest * 100
                            exit_reason = (
                                f"?몃젅?쇰쭅 怨좎젏-{TRAIL_GIVEBACK*100:.0f}% "
                                f"({give:+.2f}%, ??{net_rate*100:+.2f}%)"
                            )
                else:
                    if net_rate >= TAKE_PROFIT_CAP:
                        exit_reason = f"?듭젅 罹???{net_rate*100:.2f}% (媛寃?+{gross_rate*100:.2f}%)"

        # 4) ?쒓컙?뺣━ 30遺?(B??鍮꾪솢?깊솕)
        if exit_reason is None and trigger != "B":
            if self._now() - pos["buy_time"] >= HOLDING_TIMEOUT:
                exit_reason = f"?쒓컙?뺣━ 30遺?(??{net_rate*100:+.2f}%)"

        if exit_reason:
            self._execute_sell(stock_code, current_price, exit_reason)
