from __future__ import annotations
from dataclasses import dataclass


@dataclass
class VWAPConfig:
    # VWAP ??紐?% ?댁긽?대㈃ 媛뺤꽭濡?蹂?寃껋씤吏
    strong_zone_pct: float = 0.005  # 0.5%
    # VWAP ?댄깉 ?덉슜 (??媛믩낫?????꾨옒濡??⑥뼱吏硫??쎌꽭)
    weak_zone_pct: float = -0.003  # -0.3%
    # 理쒕? ?먯닔
    max_score: int = 5


class VWAPStrategy:
    def __init__(self, config: VWAPConfig | None = None):
        self.config = config or VWAPConfig()

    def calculate_score(self, price: float, vwap: float) -> int:
        if vwap <= 0:
            return 0
        gap = (price - vwap) / vwap

        # 媛뺥븳 ?곸듅
        if gap >= self.config.strong_zone_pct:
            return 5
        # ?곸듅
        elif gap >= 0:
            return 4
        # VWAP 洹쇱쿂
        elif gap >= self.config.weak_zone_pct:
            return 2
        # ?쎌꽭
        return 0

    def is_bullish(self, price: float, vwap: float) -> bool:
        return self.calculate_score(price, vwap) >= 4

    def evaluate(self, data: dict) -> dict:
        score = self.calculate_score(
            price=data["price"],
            vwap=data["vwap"],
        )
        return {
            "name": "VWAP",
            "score": score,
            "bullish": score >= 4,
        }


def calc_vwap(candles: list[dict]) -> float:
    """遺꾨큺 由ъ뒪?몃줈 VWAP(嫄곕옒?됯?以묓룊洹좉?) 怨꾩궛.

    VWAP = 誇(醫낃? 횞 嫄곕옒?? / 誇(嫄곕옒??

    candles: [{"close": float, "volume": float, ...}, ...] ???쒖꽌 臾닿?
             (?뱀씪 09:00遺???꾩옱源뚯????꾩껜 遺꾨큺???섍꺼???뺥솗??
              理쒓렐 N媛쒕쭔 ?섍린硫?'理쒓렐 N遺??됯퇏媛'媛 ?섏뼱 VWAP 蹂몃옒
              ?섎?(?뱀씪 ?꾩쟻 ?됯퇏)? ?щ씪吏??二쇱쓽.)

    嫄곕옒???⑹씠 0?닿굅??candles媛 鍮꾩뼱?덉쑝硫?0.0 諛섑솚
    (?몄텧遺?먯꽌 vwap<=0 ??calculate_score媛 0??泥섎━?섎?濡??덉쟾).
    """
    if not candles:
        return 0.0

    total_value = 0.0
    total_volume = 0.0
    for c in candles:
        vol = c.get("volume", 0) or 0
        close = c.get("close", 0) or 0
        if vol <= 0 or close <= 0:
            continue
        total_value += close * vol
        total_volume += vol

    if total_volume <= 0:
        return 0.0
    return total_value / total_volume
